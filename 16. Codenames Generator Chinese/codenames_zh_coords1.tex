% =====================================================================
%  Codenames — kaarten op Claude Design template
%  Kaartgrootte: 66 x 44mm (liggend), 3 kolommen x 6 rijen = 18 kaarten
%  Compileren met:  xelatex codenames_zh_coords.tex
%  (NIET met pdflatex — geen CJK-ondersteuning)
%
%  Elke kaart krijgt de karakters + pinyin TWEE keer, in hetzelfde
%  kaartvak, gecentreerd rond het midden (niet tegen de randen):
%   - onder:  normale richting, zwart
%   - boven:  180° gedraaid, grijs, schuin
%     (leesbaar voor de speler aan de overkant van de tafel)
%  Ze staan dicht bij elkaar zodat het koppel duidelijk bij dezelfde
%  kaart hoort, ook zonder zichtbare kaartrand.
% =====================================================================
\documentclass[a4paper]{article}

\usepackage[a4paper,margin=0mm]{geometry}
\usepackage{fontspec}
\usepackage{tikz}
\usepackage{xcolor}

\setmainfont{Avenir Next}                          % <- pas aan naar bv. "Times New Roman"
                                                     %    of "Helvetica Neue" op macOS
\newfontfamily\zhnormaal{BiauKaiTC}          % voorkant, rechtop
\newfontfamily\zhomgekeerd{BiauKaiTC}[FakeSlant=0.15]  % gedraaid, schuin
% Op macOS zonder Noto CJK geinstalleerd, gebruik in plaats daarvan:
%   \newfontfamily\zhnormaal{Songti TC}
%   \newfontfamily\zhomgekeerd{PingFang TC}[FakeSlant=0.15]

\pagestyle{empty}
\setlength{\parindent}{0pt}

% ---------------------------------------------------------------------
%  \Kaart{X-mm}{Y-mm}{hanzi}{pinyin}
%  X, Y = TOP-LEFT coördinaat van het kaartvak (top-left van de A4).
%  Kaartgrootte 66 x 44mm. Layout is gecentreerd i.p.v. tegen de
%  boven-/onderrand geduwd, om verwarring met de buurkaart te vermijden.
% ---------------------------------------------------------------------
\newcommand{\Kaart}[4]{%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=north west, xshift=#1mm, yshift=-#2mm, inner sep=0pt]
      at (current page.north west) {%
      \begin{minipage}[t][44mm][c]{60mm}
        \centering
        % --- Gedraaide kopie (voor lezer aan overkant tafel) ---
        \rotatebox{180}{%
          \begin{minipage}{50mm}
            \centering
            \color{gray!75}
            {\zhomgekeerd\fontsize{15}{18}\selectfont #3}\\[0.5mm]
            {\itshape\fontsize{6.5}{8}\selectfont #4}
          \end{minipage}%
        }
        \\[3.5mm]
        % --- Normale kopie ---
        \begin{minipage}{50mm}
          \centering
          \color{black}
          {\zhnormaal\fontsize{15}{18}\selectfont #3}\\[0.5mm]
          {\itshape\fontsize{6.5}{8}\selectfont #4}
        \end{minipage}
      \end{minipage}%
    };
  \end{tikzpicture}%
}
